## Why not SQL? Why not JSON?

**SQL: YES — add it.** SQL is consistently top-5 in usage surveys. Every database-heavy app has `.sql` migration and schema files. Symbol extraction maps cleanly to DDL: `CREATE TABLE`, `CREATE FUNCTION`, `CREATE VIEW`. When a query mentions "user authentication", the file defining the `users` table ranks appropriately.

**JSON: NO — skip it.** JSON is config/data with no meaningful code symbols. Key-value extraction produces noise, not signal. The YAML analyzer already handles structured config. JSON files are indexed at the file level via the fallback path — that is sufficient.

```python
"""Downstream read cost estimation for context packs.

Measures the token cost an agent pays to *read* a context pack's items.
File-pointer packs: agent reads full file (~10k tokens for 5 files).
Symbol-body packs (v4.4+): agent reads only the inlined body (~200 tokens).
"""
from __future__ import annotations
from pathlib import Path

_CHARS_PER_TOKEN = 4  # conservative estimate


def estimate_downstream_read_tokens(
    items: list[dict],  # [{"path": str, "lines": [start, end] | None}, ...]
    project_root: Path,
) -> int:
    """Return total tokens agent must read to consume all items in the pack."""
    total = 0
    for item in items:
        raw = item.get("path", "")
        lines = item.get("lines")
        path = Path(raw) if Path(raw).is_absolute() else project_root / raw
        if not path.exists():
            continue
        try:
            content = path.read_text(encoding="utf-8", errors="replace")
            all_lines = content.splitlines()
        except OSError:
            continue

        if lines and len(lines) == 2 and lines[0] and lines[1]:
            n_lines = max(0, int(lines[1]) - int(lines[0]) + 1)
        else:
            n_lines = len(all_lines)

        chars_per_line = len(content) / max(1, len(all_lines))
        total += max(1, int(n_lines * chars_per_line / _CHARS_PER_TOKEN))
    return total
```

```python
    def fetch_symbol_lines_batch(
        self,
        file_symbol_pairs: list[tuple[str, str]],
    ) -> dict[tuple[str, str], tuple[int, int]]:
        """Return {(path, name): (line_start, line_end)} for known symbols.

        Single parameterised query — no N+1 overhead regardless of pack size.
        """
        if not file_symbol_pairs:
            return {}
        placeholders = ",".join("(?,?)" for _ in file_symbol_pairs)
        flat = [v for pair in file_symbol_pairs for v in pair]
        rows = self._db.execute(
            f"SELECT file_path, name, line_start, line_end FROM symbols "
            f"WHERE (file_path, name) IN (VALUES {placeholders})",
            flat,
        ).fetchall()
        return {
            (r["file_path"], r["name"]): (r["line_start"], r["line_end"])
            for r in rows
            if r["line_start"] is not None and r["line_end"] is not None
        }
```

```python
    def _enrich_with_symbol_bodies(
        self,
        items: list[ContextItem],
        project_root: Path,
    ) -> list[ContextItem]:
        """Populate symbol_body and symbol_lines for ranked items where possible.

        Extracts the symbol name from the item title, batch-fetches line ranges
        from SymbolRepository, reads file lines from disk, stores in the item.
        Items without a resolvable symbol are returned unchanged.
        """
        from storage_sqlite.repositories import SymbolRepository

        pairs: list[tuple[str, str]] = []
        for item in items:
            if not item.path_or_ref or item.source_type in ("memory", "decision", "handover"):
                continue
            raw_name = item.title.split("(")[0].split("[")[0].strip().split()
            name = raw_name[-1] if raw_name else ""
            if name:
                pairs.append((item.path_or_ref, name))

        if not pairs:
            return items

        lines_map = SymbolRepository(self._db).fetch_symbol_lines_batch(pairs)

        result: list[ContextItem] = []
        for item in items:
            if not item.path_or_ref or item.source_type in ("memory", "decision", "handover"):
                result.append(item)
                continue
            raw_name = item.title.split("(")[0].split("[")[0].strip().split()
            name = raw_name[-1] if raw_name else ""
            key = (item.path_or_ref, name)
            if key not in lines_map:
                result.append(item)
                continue

            line_start, line_end = lines_map[key]
            file_path = (
                project_root / item.path_or_ref
                if not Path(item.path_or_ref).is_absolute()
                else Path(item.path_or_ref)
            )
            try:
                file_lines = file_path.read_text(encoding="utf-8", errors="replace").splitlines()
                body_lines = file_lines[max(0, line_start - 1): min(len(file_lines), line_end)]
                body = "\n".join(body_lines)
            except OSError:
                result.append(item)
                continue

            result.append(item.model_copy(update={
                "symbol_body": body,
                "symbol_lines": (line_start, line_end),
                "est_tokens": max(1, len(body) // 4),
            }))
        return result
```

```python
    def _apply_source_file_boost(
        self,
        items: list[ContextItem],
        query_tokens: set[str],
    ) -> list[ContextItem]:
        """Boost non-test source files whose stem matches a query token.

        When a query token (len >= 4) equals or is a prefix of a non-test
        file's stem, multiply that file's confidence by 1.3 (capped at 0.95).
        Prevents test fixtures from outranking the source file they test.
        """
        if not query_tokens or not items:
            return items
        normalised = {t.lower() for t in query_tokens if len(t) >= 4}
        result = []
        for item in items:
            new_conf = item.confidence
            path = item.path_or_ref or ""
            if path and not _is_test_or_script_path(path):
                stem = Path(path).stem.lower()
                if any(stem == qt or qt.startswith(stem) for qt in normalised):
                    new_conf = min(0.95, new_conf * 1.3)
            result.append(
                item.model_copy(update={"confidence": new_conf})
                if new_conf != item.confidence else item
            )
        return result
```

```python
"""context-router-language-go: Go language analyzer."""
from __future__ import annotations
from pathlib import Path
import tree_sitter_go as tsgo
from tree_sitter import Language, Node, Parser
from contracts.interfaces import DependencyEdge, Symbol

_LANGUAGE = Language(tsgo.language())
_PARSER = Parser(_LANGUAGE)

def _text(node: Node) -> str:
    return (node.text or b"").decode("utf-8", errors="replace")


class GoAnalyzer:
    def _is_test_file(self, file: Path) -> bool:
        return file.name.endswith("_test.go")

    def analyze(self, path: Path) -> list[Symbol | DependencyEdge]:
        try:
            source = path.read_bytes()
        except OSError:
            return []
        tree = _PARSER.parse(source)
        results: list[Symbol | DependencyEdge] = []
        self._walk(tree.root_node, path, results)
        return results

    def _walk(self, node: Node, path: Path, results: list) -> None:
        if node.type == "function_declaration":
            name_node = node.child_by_field_name("name")
            if name_node and _text(name_node) not in {"init", "main"}:
                results.append(Symbol(
                    name=_text(name_node), kind="function", file=path,
                    line_start=node.start_point[0] + 1,
                    line_end=node.end_point[0] + 1,
                    language="go", signature=f"func {_text(name_node)}(...)",
                ))
        elif node.type == "method_declaration":
            name_node = node.child_by_field_name("name")
            recv_node = node.child_by_field_name("receiver")
            if name_node:
                recv = _text(recv_node).strip("()") if recv_node else ""
                results.append(Symbol(
                    name=_text(name_node), kind="method", file=path,
                    line_start=node.start_point[0] + 1,
                    line_end=node.end_point[0] + 1,
                    language="go", signature=f"func ({recv}) {_text(name_node)}(...)",
                ))
        elif node.type == "type_declaration":
            for spec in node.children:
                if spec.type == "type_spec":
                    name_node = spec.child_by_field_name("name")
                    type_node = spec.child_by_field_name("type")
                    if name_node and type_node:
                        kind = {"struct_type": "struct", "interface_type": "interface"}.get(
                            type_node.type, "type"
                        )
                        results.append(Symbol(
                            name=_text(name_node), kind=kind, file=path,
                            line_start=node.start_point[0] + 1,
                            line_end=node.end_point[0] + 1,
                            language="go", signature=f"type {_text(name_node)} {kind}",
                        ))
        for child in node.children:
            self._walk(child, path, results)
```

```python
"""context-router-language-rust: Rust language analyzer."""
from __future__ import annotations
from pathlib import Path
import tree_sitter_rust as tsrust
from tree_sitter import Language, Node, Parser
from contracts.interfaces import DependencyEdge, Symbol

_LANGUAGE = Language(tsrust.language())
_PARSER = Parser(_LANGUAGE)

def _text(node: Node) -> str:
    return (node.text or b"").decode("utf-8", errors="replace")


class RustAnalyzer:
    def _is_test_file(self, file: Path) -> bool:
        return "tests" in file.parts or "test" in file.parts

    def analyze(self, path: Path) -> list[Symbol | DependencyEdge]:
        try:
            source = path.read_bytes()
        except OSError:
            return []
        tree = _PARSER.parse(source)
        results: list[Symbol | DependencyEdge] = []
        self._walk(tree.root_node, path, results, impl_name=None)
        return results

    def _walk(self, node: Node, path: Path, results: list, impl_name: str | None) -> None:
        if node.type == "function_item":
            name_node = node.child_by_field_name("name")
            if name_node:
                kind = "method" if impl_name else "function"
                results.append(Symbol(
                    name=_text(name_node), kind=kind, file=path,
                    line_start=node.start_point[0] + 1,
                    line_end=node.end_point[0] + 1,
                    language="rs",
                    signature=f"fn {_text(name_node)}(...)",
                ))
        elif node.type == "impl_item":
            type_node = node.child_by_field_name("type")
            impl_type = _text(type_node) if type_node else "impl"
            for child in node.children:
                self._walk(child, path, results, impl_name=impl_type)
            return
        elif node.type in ("struct_item", "enum_item", "trait_item"):
            name_node = node.child_by_field_name("name")
```

```python
"""context-router-language-sql: SQL DDL analyzer.

Uses regex on CREATE statements rather than tree-sitter because SQL dialect
fragmentation (PostgreSQL, MySQL, SQLite, T-SQL) makes grammar-based parsing
fragile. Regex DDL extraction is reliable and portable across all dialects.
"""
from __future__ import annotations
import re
from pathlib import Path
from contracts.interfaces import DependencyEdge, Symbol

_CREATE_RE = re.compile(
    r"CREATE\s+(?:OR\s+REPLACE\s+)?(?P<kind>TABLE|VIEW|FUNCTION|PROCEDURE)\s+"
    r"(?:IF\s+NOT\s+EXISTS\s+)?(?:[\w`\"]+\.)?(?P<name>[\w`\"]+)",
    re.IGNORECASE | re.MULTILINE,
)


class SqlAnalyzer:
    def _is_test_file(self, file: Path) -> bool:
        return False

    def analyze(self, path: Path) -> list[Symbol | DependencyEdge]:
        try:
            source = path.read_text(encoding="utf-8", errors="replace")
        except OSError:
            return []
        lines = source.splitlines()
        results: list[Symbol | DependencyEdge] = []
        for m in _CREATE_RE.finditer(source):
            kind = m.group("kind").lower()
            name = m.group("name").strip("`\"")
            line_start = source[: m.start()].count("\n") + 1
            next_m = _CREATE_RE.search(source, m.end())
            line_end = source[: next_m.start()].count("\n") if next_m else len(lines)
            results.append(Symbol(
                name=name, kind=kind, file=path,
                line_start=line_start, line_end=max(line_start, line_end),
                language="sql", signature=f"CREATE {kind.upper()} {name}",
            ))
        return results
```
