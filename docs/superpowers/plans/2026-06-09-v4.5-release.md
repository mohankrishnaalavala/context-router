Replace the `rglob` walk with a dir-pruned `os.walk` (the current code walks all ~45k venv files even when ignoring them) and add gitignore matching:

```python
import fnmatch
import os
import sys
from pathlib import Path
from typing import Iterator

import pathspec


class FileScanner:
    def __init__(self, root, ignore_patterns, plugin_loader):
        self._root = root
        self._ignore_patterns = ignore_patterns
        self._plugin_loader = plugin_loader
        self._gitignore = self._load_gitignore()

    def _load_gitignore(self) -> "pathspec.PathSpec | None":
        gi = self._root / ".gitignore"
        try:
            if not gi.is_file():
                return None
            lines = gi.read_text(encoding="utf-8", errors="replace").splitlines()
            return pathspec.PathSpec.from_lines("gitwildmatch", lines)
        except OSError as exc:
            print(
                f"WARN: could not read .gitignore ({exc}); "
                "falling back to ignore_patterns only",
                file=sys.stderr,
            )
            return None

    def scan(self) -> Iterator[tuple[Path, str]]:
        for dirpath, dirnames, filenames in os.walk(self._root):
            base = Path(dirpath)
            # Prune in place so we never descend into ignored trees.
            dirnames[:] = sorted(
                d for d in dirnames if not self._is_ignored(base / d, is_dir=True)
            )
            for fname in sorted(filenames):
                path = base / fname
                if self._is_ignored(path):
                    continue
                ext = path.suffix.lstrip(".")
                if not ext:
                    continue
                if self._plugin_loader.get_analyzer(ext) is not None:
                    yield path, ext

    def _is_ignored(self, path: Path, is_dir: bool = False) -> bool:
        try:
            rel = path.relative_to(self._root)
        except ValueError:
            return False
        for pattern in self._ignore_patterns:
            for part in rel.parts:
                if fnmatch.fnmatch(part, pattern):
                    return True
            if fnmatch.fnmatch(str(rel), pattern):
                return True
        if self._gitignore is not None:
            rel_str = str(rel) + ("/" if is_dir else "")
            if self._gitignore.match_file(rel_str):
                return True
        return False
```

Keep the existing class docstrings; only the bodies above change.

VENDORED_LIKE = (".venv", "venv/", "node_modules", "site-packages", "vendor/")


def _check_index_pollution(conn) -> tuple[bool, str]:
    total = conn.execute("SELECT count(*) FROM symbols").fetchone()[0]
    if total == 0:
        return True, "index empty — run 'context-router index'"
    clauses = " OR ".join("file_path LIKE ?" for _ in VENDORED_LIKE)
    vendored = conn.execute(
        f"SELECT count(*) FROM symbols WHERE {clauses}",
        tuple(f"%{v}%" for v in VENDORED_LIKE),
    ).fetchone()[0]
    pct = round(100 * vendored / total)
    if pct >= 20:

            f"index pollution: {pct}% of {total} symbols are under vendored "
            "paths — re-run 'context-router index' (v4.5+ prunes them)"
        )
    return True, f"index hygiene OK ({pct}% vendored)"

```python
#!/usr/bin/env python3
"""Judge pack sufficiency with a model: could an agent fix the task from the
pack alone? Reads a holdout score JSON, writes judged copy alongside it.

Usage: python3 benchmark/judge_packs.py <score.json> [--model claude-fable-5]
"""
from __future__ import annotations

import argparse
import json
import shutil
import subprocess
import sys
from pathlib import Path

PROMPT = """You judge context packs for AI coding agents.
Task description: {query}
Ground-truth files that must be modified: {truth}
The pack given to the agent (paths + any included bodies):
{pack}

Question: from this pack ALONE (no extra file reads), does the agent have
enough to locate and start the fix? Reply with ONLY this JSON:
{{"sufficient": true|false, "missing": "<one line: what is missing, or empty>"}}"""


def judge_one(task: dict, model: str) -> dict:
    prompt = PROMPT.format(
        query=task.get("query", task.get("task_id", "")),
        truth=task["ground_truth_files"],
        pack=json.dumps(task["predicted_files"])[:6000],
    )
    proc = subprocess.run(
        ["claude", "-p", "--model", model, prompt],
        capture_output=True, text=True, timeout=300,
    )
    raw = proc.stdout.strip()
    start, end = raw.find("{"), raw.rfind("}")
    if proc.returncode != 0 or start == -1:
        return {"sufficient": None, "missing": f"judge error: {proc.stderr[:200]}"}
    return json.loads(raw[start : end + 1])


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("score_json", type=Path)
    ap.add_argument("--model", default="claude-fable-5")
    args = ap.parse_args()

    if shutil.which("claude") is None:
        print("ERROR: claude CLI not found — judge step requires it", file=sys.stderr)
        return 1

    data = json.loads(args.score_json.read_text())
    for task in data if isinstance(data, list) else [data]:
        task["judge"] = judge_one(task, args.model)
    out = args.score_json.with_suffix(".judged.json")
    out.write_text(json.dumps(data, indent=2))
    print(f"Judged → {out}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
